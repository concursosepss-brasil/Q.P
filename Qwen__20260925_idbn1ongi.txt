    <!-- Painel do Administrador -->
    <div id="adminPanel" class="screen">
        <div class="admin-header">
            <h2>👤 Painel do Administrador</h2>
            <button class="logout-btn" onclick="adminLogout()">Sair</button>
        </div>
        <div class="stats">
            <div class="stat-card">
                <div class="stat-number" id="totalUsers">0</div>
                <div class="stat-label">Total de Usuários</div>
            </div>
            <div class="stat-card">
                <div class="stat-number" id="pendingUsers">0</div>
                <div class="stat-label">Pendentes</div>
            </div>
            <div class="stat-card">
                <div class="stat-number" id="approvedUsers">0</div>
                <div class="stat-label">Aprovados</div>
            </div>
        </div>
        <h3>Usuários Cadastrados</h3>
        <div id="loadingUsers" class="loading">
            <div class="spinner"></div>
            <p>Carregando usuários...</p>
        </div>
        <div id="userListContainer" style="display: none;">
            <ul id="userList" class="user-list"></ul>
        </div>
        <div id="emptyState" class="empty-state" style="display: none;">
            <p>Nenhum usuário cadastrado ainda</p>
        </div>
    </div>
</div>

<script>
    // =====================================================
    // ⚠️ CONFIGURAÇÃO DO FIREBASE - SUBSTITUA COM SEUS DADOS
    // =====================================================
    const firebaseConfig = {
        apiKey: "COLE_SUA_API_KEY_AQUI",
        authDomain: "SEU_PROJETO.firebaseapp.com",
        projectId: "SEU_PROJETO_ID",
        storageBucket: "SEU_PROJETO.appspot.com",
        messagingSenderId: "SEU_MESSAGING_SENDER_ID",
        appId: "SEU_APP_ID"
    };

    const ADMIN_PASSWORD = 'Atlanta';
    const USERS_COLLECTION = 'users';

    firebase.initializeApp(firebaseConfig);
    const db = firebase.firestore();

    // =====================================================
    // LÓGICA DO PAINEL ADMINISTRATIVO
    // =====================================================
    const ADMIN_SESSION_KEY = 'simulado_admin_session';

    document.addEventListener('DOMContentLoaded', function() {
        verificarSessaoAdmin();
    });

    function verificarSessaoAdmin() {
        const adminSession = localStorage.getItem(ADMIN_SESSION_KEY);
        if (adminSession === 'true') {
            mostrarTela('adminPanel');
            carregarUsuarios();
        }
    }

    function mostrarTela(screenId) {
        document.querySelectorAll('.screen').forEach(screen => screen.classList.remove('active'));
        document.getElementById(screenId).classList.add('active');
    }

    function mostrarErro(elementId, message) {
        const element = document.getElementById(elementId);
        element.textContent = message;
        element.classList.add('show');
    }

    document.getElementById('adminLoginForm').addEventListener('submit', function(e) {
        e.preventDefault();
        const password = document.getElementById('adminPassword').value;
        if (password === ADMIN_PASSWORD) {
            localStorage.setItem(ADMIN_SESSION_KEY, 'true');
            mostrarTela('adminPanel');
            carregarUsuarios();
        } else {
            mostrarErro('adminLoginError', 'Senha incorreta');
        }
    });

    async function carregarUsuarios() {
        const loadingDiv = document.getElementById('loadingUsers');
        const userListContainer = document.getElementById('userListContainer');
        const emptyState = document.getElementById('emptyState');

        try {
            const snapshot = await db.collection(USERS_COLLECTION)
                .orderBy('registeredAt', 'desc')
                .get();

            loadingDiv.style.display = 'none';

            if (snapshot.empty) {
                emptyState.style.display = 'block';
                atualizarEstatisticas(0, 0, 0);
                return;
            }

            const users = [];
            let total = 0, pending = 0, approved = 0;

            snapshot.forEach(doc => {
                const userData = doc.data();
                users.push({ id: doc.id, ...userData });
                total++;
                if (userData.approved) approved++;
                else pending++;
            });

            atualizarEstatisticas(total, pending, approved);
            renderizarListaUsuarios(users);
        } catch (error) {
            console.error('Erro ao carregar usuários:', error);
            loadingDiv.innerHTML = '<p style="color: #c33;">Erro ao carregar usuários</p>';
        }
    }

    function atualizarEstatisticas(total, pending, approved) {
        document.getElementById('totalUsers').textContent = total;
        document.getElementById('pendingUsers').textContent = pending;
        document.getElementById('approvedUsers').textContent = approved;
    }

    function renderizarListaUsuarios(users) {
        const userList = document.getElementById('userList');
        const userListContainer = document.getElementById('userListContainer');
        const emptyState = document.getElementById('emptyState');

        if (users.length === 0) {
            userListContainer.style.display = 'none';
            emptyState.style.display = 'block';
            return;
        }

        emptyState.style.display = 'none';
        userListContainer.style.display = 'block';

        userList.innerHTML = users.map(user => {
            const dataRegistro = user.registeredAt ? 
                new Date(user.registeredAt.seconds * 1000).toLocaleDateString('pt-BR') : 
                'Data não disponível';

            return `
                <li class="user-item ${user.approved ? 'approved' : ''}">
                    <div class="user-email">
                        ${user.email}
                        <div style="font-size: 12px; color: #999; font-weight: normal; margin-top: 4px;">
                            Cadastrado em: ${dataRegistro}
                        </div>
                    </div>
                    <span class="user-status ${user.approved ? 'status-approved' : 'status-pending'}">
                        ${user.approved ? '✓ Aprovado' : '⏳ Pendente'}
                    </span>
                    <div>
                        ${!user.approved ? `<button class="action-btn approve-btn" onclick="aprovarUsuario('${user.id}')">Aprovar</button>` : ''}
                        <button class="action-btn remove-btn" onclick="removerUsuario('${user.id}', '${user.email}')">Remover</button>
                    </div>
                </li>
            `;
        }).join('');
    }

    async function aprovarUsuario(userId) {
        if (!confirm('Tem certeza que deseja aprovar este usuário?')) return;
        try {
            await db.collection(USERS_COLLECTION).doc(userId).update({ approved: true });
            carregarUsuarios();
        } catch (error) {
            console.error('Erro ao aprovar usuário:', error);
            alert('Erro ao aprovar usuário. Tente novamente.');
        }
    }

    async function removerUsuario(userId, email) {
        if (!confirm(`Tem certeza que deseja remover o usuário ${email}?`)) return;
        try {
            await db.collection(USERS_COLLECTION).doc(userId).delete();
            carregarUsuarios();
        } catch (error) {
            console.error('Erro ao remover usuário:', error);
            alert('Erro ao remover usuário. Tente novamente.');
        }
    }

    function adminLogout() {
        localStorage.removeItem(ADMIN_SESSION_KEY);
        mostrarTela('adminLoginScreen');
        document.getElementById('adminLoginForm').reset();
    }
</script>
